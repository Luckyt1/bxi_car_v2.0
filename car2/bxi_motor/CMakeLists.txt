cmake_minimum_required(VERSION 3.16)
project(bxi_motor LANGUAGES CXX)

if(NOT TARGET iswv::canopen)
  set(ISWV_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ISWV" CACHE PATH
    "Path to the existing ISWV source tree")
  option(ISWV_BUILD_TESTS "Build CANopen tests" OFF)
  option(ISWV_BUILD_EXAMPLES "Build CANopen examples" OFF)
  add_subdirectory("${ISWV_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/iswv")
endif()

add_library(bxi_motor STATIC src/motor.cpp)
add_library(bxi::motor ALIAS bxi_motor)
target_compile_features(bxi_motor PUBLIC cxx_std_17)
target_include_directories(bxi_motor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(bxi_motor PUBLIC iswv::canopen)
target_compile_options(bxi_motor PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic -Wconversion -Wshadow>)

option(BXI_MOTOR_BUILD_TESTS "Build offline BXI motor protocol tests" ON)
if(BXI_MOTOR_BUILD_TESTS)
  enable_testing()
  add_executable(test_bxi_motor tests/test_bxi_motor.cpp)
  target_link_libraries(test_bxi_motor PRIVATE bxi::motor)
  add_test(NAME bxi_motor_protocol COMMAND test_bxi_motor)
  add_executable(test_bxi_motor_commands tests/test_bxi_motor_commands.cpp)
  target_link_libraries(test_bxi_motor_commands PRIVATE bxi::motor)
  add_test(NAME bxi_motor_commands COMMAND test_bxi_motor_commands)
  set_tests_properties(bxi_motor_protocol bxi_motor_commands PROPERTIES TIMEOUT 10)
endif()

option(BXI_MOTOR_BUILD_EXAMPLES "Build BXI motor offline example" ON)
if(BXI_MOTOR_BUILD_EXAMPLES)
  add_executable(bxi_motor_offline_example examples/offline.cpp)
  target_link_libraries(bxi_motor_offline_example PRIVATE bxi::motor)
  if(BXI_MOTOR_BUILD_TESTS)
    add_test(NAME bxi_motor_offline_example COMMAND bxi_motor_offline_example)
  endif()
endif()
